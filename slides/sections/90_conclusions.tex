\section{Conclusions}

\begin{frame}[t]{CAGE-NAS: Conclusions and Research Perspectives}
  \vspace{-0.13cm}
  {\fontsize{7.45}{8.9}\selectfont
  CAGE-NAS turns architecture growth into a sequence of certified decisions
  aligned with functional-gradient descent.}

  \vspace{0.10cm}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.315\textwidth}
      {\color{EnstaBlue}\bfseries\scriptsize CORE CONTRIBUTION}
      \vspace{0.04cm}

      \begin{beamercolorbox}[rounded=true,sep=0.11cm,wd=\linewidth]{demeter takeaway}
        {\fontsize{6.85}{8.2}\selectfont
        \begin{itemize}
          \setlength{\itemsep}{0.08cm}
          \item Begin with a small network and add capacity only when the
                current functional representation fails certification.
          \item Try damped tangent and nonlinear families first; if both fail,
                function-preserving growth expands the reachable directions.
          \item Use the same functional-gradient objective for training and
                for deciding \emph{when} and \emph{where} to grow.
        \end{itemize}}
      \end{beamercolorbox}
    \end{column}

    \begin{column}{0.335\textwidth}
      {\color{EnstaBlue}\bfseries\scriptsize EMPIRICAL CONCLUSION}
      \vspace{0.04cm}

      \begin{beamercolorbox}[rounded=true,sep=0.11cm,wd=\linewidth]{certified card}
        {\fontsize{6.85}{8.2}\selectfont
        \begin{itemize}
          \setlength{\itemsep}{0.09cm}
          \item Synthetic $N=1024$ runs produce stable certified trajectories
                across four seeds.
          \item On MNIST, loss decreases and accuracy rises as capacity grows.
        \end{itemize}}

        \vspace{0.05cm}
        \centering
        {\fontsize{6.35}{7.6}\selectfont\bfseries
        Controlled synthetic $N=1024$ only}\\[0.04cm]
        {\color{CertifiedGreen}\fontsize{10.5}{12.5}\selectfont\bfseries
        $0.9575\pm0.0038$}

        {\fontsize{6.25}{7.5}\selectfont four-seed mean test accuracy}\\[0.06cm]
        {\fontsize{6.55}{7.8}\selectfont
        99.83rd--100.00th percentiles; fixed retraining gives no systematic gain.}
      \end{beamercolorbox}
    \end{column}

    \begin{column}{0.315\textwidth}
      {\color{EnstaBlue}\bfseries\scriptsize LIMITATIONS AND NEXT STEPS}
      \vspace{0.04cm}

      \begin{beamercolorbox}[rounded=true,sep=0.11cm,wd=\linewidth]{growth card}
        {\fontsize{6.8}{8.1}\selectfont
        \begin{itemize}
          \setlength{\itemsep}{0.09cm}
          \item Add parameters, FLOPs, memory, and energy to a multi-objective
                growth certificate.
          \item Reduce initialization sensitivity and connect probe-level
                certification with generalization.
          \item Lower the cost of tangent representations and candidate
                evaluation at scale.
          \item Extend CAGE-NAS to CNNs and DAGs, then evaluate on CIFAR-10/100
                and NAS Unseen under consistent budgets.
        \end{itemize}}
      \end{beamercolorbox}
    \end{column}
  \end{columns}

  \vspace{0.12cm}
  \begin{beamercolorbox}[rounded=true,sep=0.09cm,wd=\linewidth]{demeter takeaway}
    \centering
    {\fontsize{7.0}{8.4}\selectfont\bfseries
    Certified functional descent can discover near-best architectures without
    exhaustive enumeration; efficiency, generalization, and scalability remain open.}
  \end{beamercolorbox}
\end{frame}
