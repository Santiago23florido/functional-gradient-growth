      \vspace{0.03cm}

      \begin{beamercolorbox}[rounded=true,sep=0.10cm,wd=\linewidth]{growth card}
        \centering
        \fontsize{6.45}{7.7}\selectfont
        \(\displaystyle
          \begin{gathered}
            s=nc,\quad p=\#\text{parameters},\quad
            G=\#\text{growth steps},\quad L=\#\text{locations},\\[4pt]
            J\in\mathbb R^{s\times p},\quad
            \operatorname{memory}(J)=\Theta(sp),\\[4pt]
            \operatorname{cost}(\text{thin SVD})
            =\Theta\!\left(sp\min\{s,p\}\right),\\[4pt]
            \operatorname{cost}(\text{grow-to-certify})
            =\Theta\!\left(GLsp\min\{s,p\}\right).
          \end{gathered}
        \)
      \end{beamercolorbox}

      \vspace{0.07cm}
      {\color{EnstaBlue}\bfseries\scriptsize ROW CHUNKING: LOWER PEAK MEMORY, SAME $J$}
      \vspace{0.03cm}

      \begin{beamercolorbox}[rounded=true,sep=0.09cm,wd=\linewidth]{demeter takeaway}
        \centering
        \fontsize{6.5}{7.8}\selectfont
        \(\displaystyle
          J=\begin{bmatrix}J_{\mathrm{row}}^{(1)};\ \cdots;\
          J_{\mathrm{row}}^{(R)}\end{bmatrix},\qquad q=256
        \)

        \vspace{0.04cm}
        Temporary activation memory falls toward $\Theta(qp)$, but the final
        $\Theta(sp)$ matrix remains and repeated forward passes add compute.
      \end{beamercolorbox}

      \vspace{0.08cm}
      {\fontsize{6.85}{8.15}\selectfont
      \begin{itemize}
        \setlength{\itemsep}{0.07cm}
        \item Matrix-free $Jv$ and $J^\top w$, factor reuse, and certification
              probes make larger runs possible.
        \item Both $s$ (probe size/output dimension) and $p$ (architecture
              size) grow, so memory and factorization time still escalate.
      \end{itemize}}
